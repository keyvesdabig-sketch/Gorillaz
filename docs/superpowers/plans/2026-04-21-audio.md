# Audio — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Chiptune-Sounddesign via Web Audio API — drei Kern-Sounds (Schuss, Explosion, Tod) synthetisiert in `audio.js`, eingebunden in `main.js`. Phase 2 fügt optionale Erweiterungen hinzu.

**Architecture:** Einzelnes `audio.js`-Modul, kein externer State. `AudioContext` wird lazy bei erstem Tastendruck initialisiert (Browser-Autoplay-Policy). Alle Sounds sind One-shot-Funktionen die eigene Nodes erzeugen und sich selbst aufräumen.

**Tech Stack:** Web Audio API (OscillatorNode, AudioBufferSourceNode, BiquadFilterNode, GainNode), Vanilla JS ES-Module

```js
export function playExplosion() {
  if (!audioCtx) return;
  const sampleRate = audioCtx.sampleRate;
  const frameCount = Math.floor(sampleRate * 0.5);
  const buffer     = audioCtx.createBuffer(1, frameCount, sampleRate);
  const data       = buffer.getChannelData(0);
  for (let i = 0; i < frameCount; i++) data[i] = Math.random() * 2 - 1;

  const source = audioCtx.createBufferSource();
  source.buffer = buffer;

  const filter = audioCtx.createBiquadFilter();
  filter.type = 'lowpass';
  const gain = audioCtx.createGain();
  const t = audioCtx.currentTime;
  filter.frequency.setValueAtTime(800, t);
  filter.frequency.linearRampToValueAtTime(100, t + 0.5);
  gain.gain.setValueAtTime(0.5, t);
  gain.gain.linearRampToValueAtTime(0, t + 0.5);

  source.connect(filter);
  filter.connect(gain);
  gain.connect(audioCtx.destination);
  source.start(t);
}
```

```js
export function playDeath() {
  if (!audioCtx) return;
  const t = audioCtx.currentTime;
  DEATH_FREQS.forEach((freq, i) => {
    const osc  = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.setValueAtTime(freq, t + i * 0.12);
    gain.gain.setValueAtTime(0.25, t + i * 0.12);
    gain.gain.setValueAtTime(0,    t + (i + 1) * 0.12);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start(t + i * 0.12);
    osc.stop(t + (i + 1) * 0.12);
  });
}
```

```js
export function playBananaFly() {
  if (!audioCtx || flyOsc) return;
  flyOsc  = audioCtx.createOscillator();
  flyGain = audioCtx.createGain();
  flyLfo  = audioCtx.createOscillator();
  const lfoGain = audioCtx.createGain();

  flyOsc.type = 'triangle';
  flyOsc.frequency.setValueAtTime(180, audioCtx.currentTime);
  flyLfo.frequency.setValueAtTime(6, audioCtx.currentTime);
  lfoGain.gain.setValueAtTime(10, audioCtx.currentTime);
  flyGain.gain.setValueAtTime(0.2, audioCtx.currentTime);

  flyLfo.connect(lfoGain);
  lfoGain.connect(flyOsc.frequency);
  flyOsc.connect(flyGain);
  flyGain.connect(audioCtx.destination);
  flyLfo.start();
  flyOsc.start();
}

export function stopBananaFly() {
  if (!flyOsc) return;
  flyOsc.stop();
  flyLfo.stop();
  flyOsc = null;
  flyLfo = null;
  flyGain = null;
}
```
